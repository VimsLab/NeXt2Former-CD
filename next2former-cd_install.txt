ubuntu 24.04

# initialize git submodules (detectron2 / Mask2Former)
git submodule update --init --recursive

# create conda env
conda create -n next2former-cd python=3.12
conda activate next2former-cd


# install pytorch 2.10.0 with CUDA12.8
pip install torch torchvision

# install cuda-toolkit 12.8
conda install nvidia::cuda-toolkit=12.8

pip install deepspeed fvcore easydict tensorboardX timm opencv-python opencv-python-headless tqdm scipy scikit-image scikit-learn python-dotenv netCDF4 torchmetrics ipykernel


conda env config vars set CUDA_HOME="$CONDA_PREFIX"
conda env config vars set LD_LIBRARY_PATH="$CONDA_PREFIX/lib64"

conda deactivate
conda activate next2former-cd

# install selective_scan
cd models/encoders/selective_scan
pip install . --no-build-isolation
cd ../../..

# install detectron2
git submodule update --init --recursive detectron2
TORCH_CUDA_ARCH_LIST="12.0" pip install -e detectron2 --no-build-isolation # "12.0" for 5090, "8.6" for a6000

# install mask2former
git submodule update --init --recursive Mask2Former

# might need to fix deprecated calls in Mask2Former/mask2former/modeling/pixel_decoder/ops/src/cuda/ms_deform_attn_cuda.cu
cd Mask2Former/mask2former/modeling/pixel_decoder/ops
TORCH_CUDA_ARCH_LIST="12.0" sh make.sh # "12.0" for 5090, "8.6" for a6000
cd ../../../../..



# Download pretrained VMamba & Dinov3 checkpoints
download Pretrained VMamba checkpoints from: 
https://github.com/MzeroMiko/VMamba
and place under pretrained/vmamba/

download Dinov3 checkpoints from: 
https://github.com/facebookresearch/dinov3
and place under pretrained/DINOv3
